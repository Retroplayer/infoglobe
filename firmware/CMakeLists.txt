cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Path to Raspberry Pi Pico SDK")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(infoglobe_lab C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(INFOGLOBE_SIG_PIN 2 CACHE STRING "GPIO connected to carrier module SIG")
if(NOT INFOGLOBE_SIG_PIN MATCHES "^[0-9]+$" OR INFOGLOBE_SIG_PIN GREATER 29)
    message(FATAL_ERROR "INFOGLOBE_SIG_PIN must be 0..29")
endif()
pico_sdk_init()
add_executable(infoglobe_lab main.c console.c)
target_compile_definitions(infoglobe_lab PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 INFOGLOBE_SIG_PIN=${INFOGLOBE_SIG_PIN})
target_compile_options(infoglobe_lab PRIVATE -Wall -Wextra -Werror)
target_link_libraries(infoglobe_lab pico_stdlib hardware_pio)
pico_generate_pio_header(infoglobe_lab ${CMAKE_CURRENT_LIST_DIR}/transmit.pio)
pico_enable_stdio_usb(infoglobe_lab 1)
pico_enable_stdio_uart(infoglobe_lab 0)
pico_add_extra_outputs(infoglobe_lab)

set(INFOGLOBE_RX_PIN 3 CACHE STRING "Receiver DAT GPIO")
if(NOT INFOGLOBE_RX_PIN MATCHES "^[0-9]+$" OR INFOGLOBE_RX_PIN GREATER 29)
    message(FATAL_ERROR "INFOGLOBE_RX_PIN must be 0..29")
endif()
add_executable(infoglobe_capture recorder/main.c recorder/usb_descriptors.c)
target_include_directories(infoglobe_capture PRIVATE ${CMAKE_CURRENT_LIST_DIR}/recorder)
target_compile_definitions(infoglobe_capture PRIVATE INFOGLOBE_RX_PIN=${INFOGLOBE_RX_PIN})
target_compile_options(infoglobe_capture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(infoglobe_capture pico_stdlib pico_unique_id hardware_pio hardware_dma hardware_irq tinyusb_device)
pico_generate_pio_header(infoglobe_capture ${CMAKE_CURRENT_LIST_DIR}/recorder/capture.pio)
pico_add_extra_outputs(infoglobe_capture)
